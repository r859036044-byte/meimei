    AudioStreamPlayer _dailyVoice;
    void StopDailyVoice() { _dailyVoice?.Stop(); }
    void PlayDailyVoice(string speaker, string text)
    {
        StopDailyVoice();
        if (IsFast || speaker != "莉奈" || text == null || !DailyVoiceLines.TryGetValue(text, out var clip)) return;
        var stream = GD.Load<AudioStream>("res://Resources/Audio/Voice/RJ01047725/" + clip + ".wav");
        if (stream == null) return;
        if (_dailyVoice == null)
        {
            _dailyVoice = new AudioStreamPlayer { Name = "DailyDialogueVoice", MaxPolyphony = 1 };
            AddChild(_dailyVoice);
        }
        _dailyVoice.VolumeLinear = SettingsManager.Instance?.VoiceVolume ?? 1f;
        _dailyVoice.StreamPaused = false;
        _dailyVoice.Stream = stream;
        _dailyVoice.Play();
    }
    void UpdateDailyVoice()
    {
        if (_dailyVoice == null) return;
        if (IsFast || !IsInstanceValid(_box) || !_box.IsVisibleInTree()) { StopDailyVoice(); return; }
        _dailyVoice.VolumeLinear = SettingsManager.Instance?.VoiceVolume ?? 1f;
        _dailyVoice.StreamPaused = Paused;
    }
